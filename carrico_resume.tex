% Custom commands
%
% Layout model for one entry:
%   \heading{Name}{Location}      bold left / plain right, starts the entry
%   \subtitle{Title}{Date range}  italic left / italic right, optional, no
%                                 new bullet -- renders directly under \heading
%   \begin{bullets} \item ... \end{bullets}   optional detail bullets

% Each transition's spacing is owned by exactly one place: the gap between
% entries lives on \heading's leading vspace, the gap between a heading and
% its subtitle lives on \subtitle's leading vspace, and the gap before the
% detail bullets lives on \bullets' leading vspace.
\newenvironment{entries}
  {\begin{itemize}[leftmargin=0.15in, label={}, topsep=0pt, parsep=0pt, itemsep=0pt]}
  {\end{itemize}}

\newenvironment{bullets}
  {\vspace{-10pt}\begin{itemize}[topsep=0pt, partopsep=0pt]\small}
  {\end{itemize}}
